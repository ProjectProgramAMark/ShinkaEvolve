import jax.numpy as jnp
# EVOLVE-BLOCK-START
def make_offspring(
    parent_genome_summary,
    parent_stats,
    population_stats,
    operator_stats,
    rng,
):
    """Evidence-weighted mutation-budget scheduler for six heredity actions."""
    success = operator_stats[0]
    usage = operator_stats[1]
    evidence = operator_stats[2]
    node_fraction = parent_genome_summary[0]
    connection_fraction = parent_genome_summary[1]
    energy_fraction = parent_stats[0]
    intake_ema = parent_stats[1]
    age_fraction = parent_stats[2]
    alive_fraction = population_stats[0]
    mean_energy_fraction = population_stats[1]
    population_change_ema = population_stats[2]
    birth_rate_ema = population_stats[3]
    death_rate_ema = population_stats[4]
    mean_intake_ema = population_stats[5]
    decline = jnp.clip(-population_change_ema, 0.0, 1.0)
    growth = jnp.clip(population_change_ema, 0.0, 1.0)
    death = jnp.clip(death_rate_ema, 0.0, 1.0)
    low_alive = jnp.clip(1.0 - alive_fraction, 0.0, 1.0)
    energy_gap = jnp.clip(0.58 - mean_energy_fraction, 0.0, 1.0)
    intake_gap = jnp.clip(0.56 - mean_intake_ema, 0.0, 1.0)
    birth_gap = jnp.clip(0.30 - birth_rate_ema, 0.0, 1.0)
    compact = jnp.clip(1.0 - 0.42 * node_fraction - 0.58 * connection_fraction, 0.0, 1.0)
    parent_vigor = jnp.clip(
        0.54 * energy_fraction + 0.34 * intake_ema + 0.12 * (1.0 - age_fraction),
        0.0,
        1.0,
    )
    reproduction_ready = jnp.clip(
        parent_vigor
        * (0.50 + 0.50 * energy_fraction)
        * (0.44 + 0.56 * intake_ema)
        * (0.56 + 0.44 * birth_rate_ema),
        0.0,
        1.0,
    )
    stress = jnp.clip(
        0.54 * decline
        + 0.46 * death
        + 0.22 * low_alive
        + 0.28 * intake_gap
        + 0.20 * energy_gap
        + 0.10 * birth_gap,
        0.0,
        1.0,
    )
    injury_signal = jnp.clip(
        (0.62 * decline + 0.54 * death + 0.18 * low_alive)
        * (0.46 + 0.34 * intake_gap + 0.20 * energy_gap),
        0.0,
        1.0,
    )
    stable = jnp.clip(
        alive_fraction
        * (1.0 - decline)
        * (1.0 - death)
        * (0.50 + 0.50 * mean_energy_fraction)
        * (0.62 + 0.38 * mean_intake_ema),
        0.0,
        1.0,
    )
    prior = jnp.array([0.36, 0.50, 0.34, 0.12, 0.10, 0.18], dtype=jnp.float32)
    learned = jnp.clip(evidence * success + (1.0 - evidence) * prior, 0.0, 1.0)
    confidence = jnp.clip(0.16 + 0.84 * evidence, 0.0, 1.0)
    underused = jnp.clip((1.0 - usage) * (1.0 - 0.50 * evidence), 0.0, 1.0)
    clone_floor = jnp.clip(0.30 + 0.34 * evidence[0] * success[0], 0.0, 1.0)
    clone_weak = jnp.clip(clone_floor - learned[0] + 0.10 * evidence[0] * usage[0], 0.0, 1.0)
    edge = jnp.clip((learned - learned[0] + 0.055) * confidence, 0.0, 1.0)
    stress_window = jnp.clip(injury_signal * (0.38 + 0.62 * stress) * reproduction_ready, 0.0, 1.0)
    stable_window = jnp.clip(stable * (1.0 - injury_signal) * reproduction_ready, 0.0, 1.0)
    tail_guard = jnp.clip((1.0 - 0.66 * low_alive) * (0.58 + 0.42 * compact), 0.0, 1.0)
    base_trickle = jnp.clip(
        stable_window
        * (0.20 + 0.80 * edge[1])
        * (0.55 + 0.45 * clone_weak)
        * (1.0 - 0.52 * growth),
        0.0,
        1.0,
    )
    rescue_budget = jnp.clip(
        stress_window
        * tail_guard
        * (0.28 + 0.72 * clone_weak)
        * (0.35 + 0.65 * jnp.clip(edge[1] + edge[2], 0.0, 1.0)),
        0.0,
        1.0,
    )
    con_score = jnp.clip(
        0.72 * edge[1]
        + 0.18 * underused[1]
        + 0.10 * compact
        + 0.16 * base_trickle
        + 0.10 * stress_window,
        0.0,
        1.0,
    )
    std_score = jnp.clip(
        0.78 * edge[2]
        + 0.16 * underused[2] * stress_window
        + 0.18 * stress
        + 0.12 * injury_signal
        - 0.18 * stable,
        0.0,
        1.0,
    )
    mix_score = jnp.clip(
        0.58 * edge[5]
        + 0.16 * underused[5] * stress_window
        + 0.10 * compact
        - 0.12 * stable,
        0.0,
        1.0,
    )
    exp_score = jnp.clip(
        0.46 * edge[3]
        + 0.10 * underused[3] * stress_window
        + 0.06 * compact
        - 0.18 * stable,
        0.0,
        1.0,
    )
    struct_score = jnp.clip(
        0.42 * edge[4]
        + 0.08 * underused[4] * stress_window
        + 0.08 * compact
        - 0.20 * stable,
        0.0,
        1.0,
    )
    conservative_gate = jnp.clip(
        0.52 * base_trickle
        + 0.46 * rescue_budget * con_score
        + 0.10 * stable_window * edge[1] * clone_weak,
        0.0,
        1.0,
    )
    standard_gate = jnp.clip(
        rescue_budget
        * std_score
        * (0.44 + 0.56 * underused[2])
        * (1.0 - 0.78 * stable * (1.0 - injury_signal)),
        0.0,
        1.0,
    )
    mixed_gate = jnp.clip(
        0.32 * rescue_budget
        * mix_score
        * compact
        * (1.0 - 0.80 * stable),
        0.0,
        1.0,
    )
    exploratory_gate = jnp.clip(
        0.16 * rescue_budget
        * exp_score
        * compact
        * (1.0 - 0.86 * stable),
        0.0,
        1.0,
    )
    structural_gate = jnp.clip(
        0.10 * rescue_budget
        * struct_score
        * compact
        * (1.0 - 0.88 * stable),
        0.0,
        1.0,
    )
    stable_clone_bonus = jnp.clip(
        stable * (1.0 - injury_signal) * (0.92 + 0.08 * growth) * (1.0 - 0.22 * clone_weak),
        0.0,
        1.0,
    )
    rescue_clone_release = jnp.clip(
        rescue_budget
        + 0.55 * standard_gate
        + 0.22 * conservative_gate
        + 0.16 * mixed_gate,
        0.0,
        1.0,
    )
    logits = jnp.array(
        [
            4.26 + 1.34 * stable_clone_bonus - 2.28 * rescue_clone_release - 0.42 * clone_weak,
            -1.34 + 2.18 * conservative_gate + 0.54 * con_score - 0.84 * stable_clone_bonus,
            -3.58 + 3.34 * standard_gate + 1.28 * rescue_budget + 0.82 * std_score - 1.02 * stable_clone_bonus,
            -6.34 + 0.86 * exploratory_gate + 0.22 * exp_score - 0.42 * stable_clone_bonus,
            -6.58 + 0.68 * structural_gate + 0.18 * struct_score - 0.46 * stable_clone_bonus,
            -6.10 + 1.24 * mixed_gate + 0.30 * mix_score - 0.38 * stable_clone_bonus,
        ],
        dtype=jnp.float32,
    )
    return jnp.clip(logits, -7.0, 7.0)
